% SLIDE 10 — OVERALL INTERPRETATION

\begin{frame}{Overall Interpretation}

\vspace{0.15cm}

{\large\color{mitoblue}\textbf{Main Findings}}

\vspace{0.30cm}

\begin{itemize}

    \setlength{\itemsep}{0.32cm}

    \item
    PCR-induced chimeric reads can be detected using both
    \textbf{alignment-derived features} and
    \textbf{raw nucleotide sequences}.

    \item
    \textbf{CNN1D achieved the strongest overall classification performance},
    with an F1 score of \textbf{0.912} and ROC-AUC of \textbf{0.961}.

    \item
    Chimera filtering can substantially reduce
    \textbf{assembly graph complexity}, but its effectiveness
    depends on \textbf{available read depth}.

\end{itemize}

\begin{block}{Gradient Boosting may be safer}

    \setlength{\itemsep}{6pt}

    \item Higher precision
    \item Preserves more clean reads
    \item Maintains assembly support

\end{itemize}

\end{block}

\end{column}

% -------------------------

\begin{column}{0.48\textwidth}

\vspace{0.18cm}

\begin{block}{CNN1D may be more advantageous}

\small

\begin{itemize}
    \setlength{\itemsep}{6pt}

    \item Higher recall
    \item Removes more chimeras
    \item Remaining reads can compensate

\end{itemize}

\end{block}

\end{column}

    \item
    Users can select between
    \textbf{Gradient Boosting} and \textbf{CNN1D},
    as well as adjust the
    \textbf{filtering threshold},
    depending on the characteristics of their dataset.

\end{itemize}

\end{frame}

\begin{frame}{Limitations}

\vspace{0.35cm}

\begin{itemize}

    \setlength{\itemsep}{0.30cm}

    \item
    The study used \textbf{simulated sequencing reads}.

    \item
    Evaluation was performed using a
    \textbf{single mitochondrial reference} from
    \textit{Sardinella lemuru}.

    \item
    Simulation cannot fully reproduce real-world sequencing conditions,
    including:

    \begin{itemize}
        \setlength{\itemsep}{0.10cm}

        \item Uneven coverage
        \item Real sequencing errors
        \item Contaminants
        \item Nuclear mitochondrial sequences
        \item Library-specific artifacts
        \item Differences among taxa and laboratory protocols

    \end{itemize}

\end{itemize}
